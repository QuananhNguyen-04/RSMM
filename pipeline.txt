# ===============================
# PIPELINE OVERVIEW
# ===============================

CONFIG: config.yaml

[STEP 1] Automatic Speech Recognition (ASR)
------------------------------------------
File: new_demo_eng.py
Input:
  - paths.audio_files
Output:
  - transcriptions/*.json

Command:
  python new_demo_eng.py 

[STEP 2] Summarization
------------------------------------------
File: summarization.py
Input:
  - paths.transcriptions.input_1
  - paths.transcriptions.input_2
Output:
  - summarized results

Command:
  python summarization.py 


[STEP 3] Dataset Processing
------------------------------------------
File: test_set_process.py
Input:
  - paths.dataset.words_dir
  - paths.dataset.meeting_id
Output:
  - processed dataset JSON

Command:
  python test_set_process.py 

[STEP 4] Evaluation
------------------------------------------
File: measurement1.py
Input:
  - paths.evaluation.ground_truth
  - paths.evaluation.prediction
Output:
  - evaluation metrics (DER, etc.)

Command:
  python measurement1.py 

[STEP 5] Summarization Evaluation
------------------------------------------
File: measurement2.py
Input:
  paths.evaluation.ground_truth
  paths.evaluation.speaker_summary
  paths.evaluation.topic_summary
  paths.evaluation.meeting_summary

Metrics:
  ICS (Information Coverage Score)
  CR (Compression Ratio)
  HR (Hallucination Rate)
  ROUGE-L

Output:
  summary evaluation metrics
  detailed preservation / hallucination analysis

Command:
  python measurement2.py
